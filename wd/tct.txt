==================================================
STEP 3 — ADD MENU TO index.html
==================================================

<div class="menu">

    <button class="menu-btn"
        onclick="navigatePage(this, 'home.html')">

        Home

    </button>

    <button class="menu-btn"
        onclick="navigatePage(this, 'about.html')">

        About

    </button>

    <button class="menu-btn"
        onclick="navigatePage(this, 'contact.html')">

        Contact

    </button>

</div>

==================================================
STEP 4 — ADD iframe
==================================================

<iframe id="contentFrame"
        src="home.html">
</iframe>

==================================================
STEP 5 — ADD CSS INSIDE <head>
==================================================

<style>

    body {
        margin: 0;
        font-family: Arial;
    }

    .menu {

        display: flex;
        flex-direction: row;

        gap: 10px;

        padding: 10px;

        background-color: #333;
    }

    .menu button {

        padding: 10px 20px;

        border: none;

        background-color: #555;

        color: white;

        cursor: pointer;

        border-radius: 5px;

        transition: 0.3s;
    }

    .menu button:hover {

        background-color: #777;
    }

    .menu button.selected {

        background-color: orange;

        color: black;

        font-weight: bold;
    }

    iframe {

        width: 100%;

        height: calc(100vh - 60px);

        border: none;
    }

</style>

==================================================
STEP 6 — ADD JAVASCRIPT BEFORE </body>
==================================================

<script>

    function navigatePage(buttonElement, url)
    {
        const buttons =
            document.querySelectorAll(".menu-btn");

        buttons.forEach(btn =>
        {
            btn.classList.remove("selected");
        });

        buttonElement.classList.add("selected");

        document.getElementById("contentFrame").src
            = url;
    }

</script>
