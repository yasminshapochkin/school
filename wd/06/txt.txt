        document.addEventListener("DOMContentLoaded", () => {

            const output = document.querySelector("#output");

            function print(text){
                output.value += text + "\n";
                console.log(text);
            }

            // ===============================
            // let / const
            // ===============================

            document.querySelector("#btnVariables")
                .addEventListener("click", () => {

                print("===== let / const =====");

                let age = 25;
                age = 30;

                const country = "Israel";

                print("Age: " + age);
                print("Country: " + country);
                print("");
            });

            // ===============================
            // Basic Types
            // ===============================

            document.querySelector("#btnTypes")
                .addEventListener("click", () => {

                print("===== Basic Types =====");

                let name = "David";
                let number = 100;
                let isOnline = true;
                let emptyValue = null;
                let data;

                print(typeof name);
                print(typeof number);
                print(typeof isOnline);
                print(typeof emptyValue);
                print(typeof data);
                print("");
            });

            // ===============================
            // == VS ===
            // ===============================

            document.querySelector("#btnCompare")
                .addEventListener("click", () => {

                print("===== == VS === =====");

                print("'5' == 5 : " + ('5' == 5));
                print("'5' === 5 : " + ('5' === 5));

                print("true == 1 : " + (true == 1));
                print("true === 1 : " + (true === 1));

                print("");
            });

            // ===============================
            // String Operations
            // ===============================

            document.querySelector("#btnStrings")
                .addEventListener("click", () => {

                print("===== String Operations =====");

                let first = "Hello";
                let second = "World";

                let full = first + " " + second;

                print(full);

                print(full.toUpperCase());

                print("Length: " + full.length);

                print("Contains 'World': " + full.includes("World"));

                print("");
            });

            // ===============================
            // Arrays
            // ===============================

            document.querySelector("#btnArrays")
                .addEventListener("click", () => {

                print("===== Array Operations =====");

                let colors = ["Red", "Green", "Blue"];

                print(colors.join(", "));

                colors.push("Yellow");

                print("After push:");
                print(colors.join(", "));

                colors.pop();

                print("After pop:");
                print(colors.join(", "));

                print("Index 1: " + colors[1]);

                print("");
            });

            // ===============================
            // Loops
            // ===============================

            document.querySelector("#btnLoops")
                .addEventListener("click", () => {

                print("===== Loops =====");

                print("FOR LOOP");

                for(let i = 1; i <= 5; i++){
                    print("i = " + i);
                }

                print("WHILE LOOP");

                let counter = 1;

                while(counter <= 3){
                    print("counter = " + counter);
                    counter++;
                }

                print("");
            });

            // ===============================
            // Methods
            // ===============================

            document.querySelector("#btnMethods")
                .addEventListener("click", () => {

                print("===== Methods =====");

                function add(a, b){
                    return a + b;
                }

                function greet(name){
                    return "Hello " + name;
                }

                print(add(5, 10));

                print(greet("Gad"));

                print("");
            });

            // ===============================
            // Arrow Functions
            // ===============================

            document.querySelector("#btnArrow")
                .addEventListener("click", () => {

                print("===== Arrow Functions =====");

                const multiply = (a, b) => {
                    return a * b;
                };

                const sayHi = name => "Hi " + name;

                print(multiply(4, 5));

                print(sayHi("Student"));

                print("");
            });

            // ===============================
            // JSON Objects
            // ===============================

            document.querySelector("#btnObjects")
                .addEventListener("click", () => {

                print("===== JSON Objects =====");

                const student = {
                    id: 1,
                    name: "Dana",
                    grade: 95,
                    isActive: true
                };

                print(student.name);
                print(student.grade);

                const jsonText = JSON.stringify(student);

                print(jsonText);

                const newObject = JSON.parse(jsonText);

                print(newObject.name);

                print("");
            });

            // ===============================
            // Clear
            // ===============================

            document.querySelector("#btnClear")
                .addEventListener("click", () => {

                output.value = "";
                console.clear();
            });

        });
